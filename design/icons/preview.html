単一HTMLアプリへの貼り方

アイコンをクリックすると、そのSVGがコピーされます。そのままHTMLに貼れます。
まとめて使うなら「ICONS まとめてコピー」を押して、下の形で使うのが楽です。 

<script>
// ↑「ICONS まとめてコピー」で出てくるのがこれ
const ICONS = { moon: '<svg …>…</svg>', timer: '<svg …>…</svg>' };

// 使うとき
function icon(name, size = 20){
  const s = ICONS[name];
  if (!s) return '';
  return s.replace('<svg', `<svg width="${size}" height="${size}"`);
}
el.innerHTML = icon('moon', 24) + 'おやすみ';
</script>

<style>
/* 色は CSS の color で決まります(SVG の中は currentColor のまま) */
.tabbar button{ color: var(--muted) }
.tabbar button.on{ color: var(--primary) }
</style>
